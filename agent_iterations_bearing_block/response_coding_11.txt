```python
from build123d import *
import math

# --- Base (centered at origin, Z from -5 to +5) ---
base = Box(100, 30, 10)

# --- Cylindrical Housing ---
# Housing from Z=+5 to Z=+45 => center at Z = +25
housing_outer = Cylinder(25, 40).translate((0, 0, 25))

# --- Union base + housing as a single solid ---
pillow_block = base + housing_outer

# --- Torus-based fillet at junction ---
fillet_r = 5
# Torus major radius = 25 + fillet_r, minor radius = fillet_r
# Positioned so tube center ring is at (r=30, Z=10)
torus = Torus(25 + fillet_r, fillet_r).translate((0, 0, 5 + fillet_r))

# Cut away everything except the concave-corner-filling quarter
inner_cut = Cylinder(25 + fillet_r, 60)
upper_cut = Box(200, 200, 20).translate((0, 0, 5 + fillet_r + 10))
fillet_piece = torus - inner_cut - upper_cut

# Union with pillow block
pillow_block = pillow_block + fillet_piece

# --- Central bore ---
bore = Cylinder(15, 52).translate((0, 0, 20))

# --- Mounting holes ---
hole1 = Cylinder(5, 12).translate(( 40, 0, 0))
hole2 = Cylinder(5, 12).translate((-40, 0, 0))

# --- Final subtraction ---
pillow_block = pillow_block - bore - hole1 - hole2

# Ensure OBJ is a single Solid, not a ShapeList
if isinstance(pillow_block, ShapeList):
    OBJ = pillow_block[0]
else:
    OBJ = pillow_block
```
